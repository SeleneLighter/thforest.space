设 为可逆矩阵, 考虑如何利用 Gauss 消去法或 LU 分解计算 的逆矩阵, 并实现相应算法. 将计算结果与MATLAB、Octave、北太天元等软件中的内置线性方程求解函数或 inv 函数进行比较, 以验证算法的正确性.

Proof

算法的实现:

import numpy as np


def lu(A):
    A = A.astype(float)
    n = A.shape[0]
    L = np.eye(n)
    U = A.copy()
    for k in range(n - 1):
        for i in range(k + 1, n):
            L[i, k] = U[i, k] / U[k, k]
            U[i, k:] -= L[i, k] * U[k, k:]
    return L, U


def f_calc(L, b):
    n = len(b)
    x = np.zeros(n)
    for i in range(n):
        x[i] = b[i] - L[i, :i] @ x[:i]
    return x


def b_calc(U, b):
    n = len(b)
    x = np.zeros(n)
    for i in range(n - 1, -1, -1):
        x[i] = (b[i] - U[i, i + 1:] @ x[i + 1:]) / U[i, i]
    return x


def inverse_mat(A):
    n = A.shape[0]
    L, U = lu(A)
    A_inv = np.zeros_like(A)
    I = np.eye(n)
    for j in range(n):
        y = f_calc(L, I[:, j])
        x = b_calc(U, y)
        A_inv[:, j] = x
    return A_inv
